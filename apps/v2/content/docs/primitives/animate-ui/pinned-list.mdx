## Usage

```tsx
import * as React from 'react';
import {
  PinnedList,
  PinnedListItem,
  PinnedListItems,
  PinnedListLabel,
  PinnedListPinned,
  PinnedListTrigger,
  PinnedListUnpinned,
} from '@animate-ui/react/primitives/pinned-list';

export function Folders() {
  const [items, setItems] = React.useState(ITEMS);
  const toggle = (id: string) =>
    setItems((prev) => prev.map((i) => (i.id === id ? { ...i, pinned: !i.pinned } : i)));

  const pinned = items.filter((i) => i.pinned);
  const unpinned = items.filter((i) => !i.pinned);

  return (
    <PinnedList onPinnedChange={toggle}>
      <PinnedListPinned>
        <PinnedListLabel hide={pinned.length === 0}>Pinned</PinnedListLabel>
        <PinnedListItems>
          {pinned.map((item) => (
            <PinnedListItem key={item.id} id={item.id} customTrigger>
              {item.name}
              <PinnedListTrigger aria-label={`Unpin ${item.name}`}>…</PinnedListTrigger>
            </PinnedListItem>
          ))}
        </PinnedListItems>
      </PinnedListPinned>
      <PinnedListUnpinned>
        <PinnedListLabel hide={unpinned.length === 0}>All</PinnedListLabel>
        <PinnedListItems>
          {unpinned.map((item) => (
            <PinnedListItem key={item.id} id={item.id} customTrigger>
              {item.name}
              <PinnedListTrigger aria-label={`Pin ${item.name}`}>…</PinnedListTrigger>
            </PinnedListItem>
          ))}
        </PinnedListItems>
      </PinnedListUnpinned>
    </PinnedList>
  );
}
```

You own the pinned state: `onPinnedChange` tells you which item to move. Without `customTrigger`, clicking the whole item toggles it.

## Animation

- The item glides to its new place on a spring (a shared layout animation), above the other items while it moves. The others make room.
- Labels fade in and out when a group becomes empty or filled (600ms).
- Items and triggers respond to hover (scale 1.02) and press (scale 0.96) in 150ms.
- `<AnimateUIProvider tempo>` paces every transition.
- Under reduced motion, items jump to their new place and don't scale.
